% The line l(t) = A + t v: the parameter t measures how many copies of v one walks from A.
\begin{tikzpicture}[every node/.style={font=\small}, >={Latex[length=2mm]}]
	\draw[->] (-1.2,0) -- (6.2,0) node[below right] {$x_1$};
	\draw[->] (0,-0.8) -- (0,3.6) node[above left] {$x_2$};
	\draw[blue, thick] (-1.0,-0.4) -- (6.0,3.1);
	\node[blue, anchor=west] at (6.0,3.25) {$\ell$};
	\draw[->, gray, thick] (0,0) -- (1,0.6);
	\node[gray, font=\small] at (0.25,0.5) {$A$};
	\draw[->, red, very thick] (1,0.6) -- (2.6,1.4) node[midway, below right, font=\small] {$v$};
	\foreach \t/\x/\y/\p in {-1/-0.6/-0.2/below, 0/1/0.6/below, 1/2.6/1.4/below, 2/4.2/2.2/below} {
		\fill[blue] (\x,\y) circle (2pt);
		\node[blue, \p=4pt, font=\small] at (\x,\y) {$\ell(\t)$};
	}
	\node[anchor=north, align=flush center, text width=10cm] at (2.5,-0.9) {$\ell(t)=A+tv$, $t\in\mathbb{R}$, with $A=\ell(0)$: equally spaced $t$ give equally spaced points. The same line has many representations (another point on it, another multiple of $v$).};
\end{tikzpicture}
