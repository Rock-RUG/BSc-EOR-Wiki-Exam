% exactly one of the B_i happens.
\begin{tikzpicture}[every node/.style={font=\small}]
	\fill[blue!10] (-3,-1.8) rectangle (-1.2,1.8);
	\fill[red!10] (-1.2,-1.8) rectangle (0.3,1.8);
	\fill[green!12] (0.3,-1.8) rectangle (1.6,1.8);
	\fill[orange!15] (1.6,-1.8) rectangle (3,1.8);
	\draw[thick] (-3,-1.8) rectangle (3,1.8);
	\draw[thick] (-1.2,-1.8) -- (-1.2,1.8);
	\draw[thick] (0.3,-1.8) -- (0.3,1.8);
	\draw[thick] (1.6,-1.8) -- (1.6,1.8);
	\node at (-2.1,0) {$B_1$};
	\node at (-0.45,0) {$B_2$};
	\node at (0.95,0) {$B_3$};
	\node at (2.3,0) {$B_4$};
	\node[anchor=south west] at (-3,1.8) {$S$};
	\node[anchor=north, align=flush center, text width=10cm] at (0,-2.0) {$B_i\cap B_j=\varnothing$ for $i\neq j$ and $B_1\cup\cdots\cup B_4=S$, so $\mathbb{P}(B_1)+\cdots+\mathbb{P}(B_4)=1$; a partition splits any question into cases: $\mathbb{P}(A)=\sum_i \mathbb{P}(A\cap B_i)$ (law of total probability)};
\end{tikzpicture}
